DESIGN (decided — follow it)
  New module asreview/database/fork_migrations.py:

    """Fork-owned schema migrations (Spec 0001, D8).

    `PRAGMA user_version` belongs to upstream ASReview, so fork schema
    changes are tracked in their own ledger table, `fork_migrations`.
    Each revision is applied once, in order, inside BEGIN IMMEDIATE so
    that concurrent processes opening the same database cannot apply it
    twice. Revision ids follow Alembic style ("0001_<slug>") to keep a
    later move to Alembic mechanical.
    """

    def _rev_0001_change_tracking(cur, project_id):
        cur.execute("""CREATE TABLE IF NOT EXISTS project_state (
            project_id TEXT PRIMARY KEY,
            version INTEGER NOT NULL DEFAULT 0,
            config_version INTEGER NOT NULL DEFAULT 0,
            pruned_version INTEGER NOT NULL DEFAULT 0,
            last_pruned_at REAL)""")
        cur.execute("""CREATE TABLE IF NOT EXISTS record_versions (
            project_id TEXT NOT NULL,
            record_id INTEGER NOT NULL,
            version INTEGER NOT NULL,
            PRIMARY KEY (project_id, record_id))""")
        cur.execute("""CREATE TABLE IF NOT EXISTS change_log (
            project_id TEXT NOT NULL,
            version INTEGER NOT NULL,
            record_id INTEGER NOT NULL,
            source TEXT NOT NULL,
            changed_at REAL NOT NULL,
            PRIMARY KEY (project_id, version))""")
        cur.execute("""CREATE INDEX IF NOT EXISTS idx_change_log_changed_at
            ON change_log (project_id, changed_at)""")
        cur.execute(
            "INSERT OR IGNORE INTO project_state (project_id) VALUES (?)",
            (project_id,))

    REVISIONS = [
        ("0001_change_tracking", _rev_0001_change_tracking),
    ]

    def apply_fork_migrations(conn, project_id):
        """Apply missing revisions; return the list of ids applied now."""
        Steps:
        1. CREATE TABLE IF NOT EXISTS fork_migrations (
               revision TEXT PRIMARY KEY, applied_at REAL NOT NULL); commit.
        2. Fast path: applied = {r for (r,) in SELECT revision FROM
           fork_migrations}. If every REVISIONS id is in it → return [].
        3. Slow path:
             conn.commit()                    # end any implicit txn
             conn.execute("BEGIN IMMEDIATE")  # waits up to the busy timeout
             try:
                 re-read `applied` (another process may have finished)
                 cur = conn.cursor()
                 for rev_id, fn in REVISIONS:
                     if rev_id in applied: continue
                     fn(cur, project_id)
                     cur.execute("INSERT INTO fork_migrations
                                  (revision, applied_at) VALUES (?, ?)",
                                 (rev_id, time.time()))
                     done.append(rev_id)
                 conn.commit()
             except BaseException:
                 conn.rollback(); raise
             return done
     Use `logging.info("Applied fork migration %s", rev_id)` per revision.
